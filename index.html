<script>
// Elements
const aboutButton = document.getElementById('about-btn');
const contactInfo = document.getElementById('contact-info');
const backButton = document.getElementById('back-btn');
const homeView = document.getElementById('home-view');
const aboutView = document.getElementById('about-view');
const footerText = document.querySelector('.footer-text');
const wipNote = document.getElementById('wip-note');
const newsletter = document.getElementById('newsletter');
const newsletterButton = document.getElementById('newsletter-btn');
const newsletterForm = document.getElementById('newsletter-form');
const newsletterEmail = document.getElementById('newsletter-email');
const newsletterConsent = document.getElementById('newsletter-consent');
const newsletterSubmit = document.getElementById('newsletter-submit');
const newsletterSpinner = document.getElementById('newsletter-spinner');
const newsletterSubmitLabel = document.getElementById('newsletter-submit-label');
let isSending = false;
// Show a spinner on the button and lock the form while the signup is being sent
function setSending(sending) {
isSending = sending;
newsletterSubmit.disabled = sending;
newsletterSubmit.classList.toggle('is-loading', sending);
newsletterSubmit.setAttribute('aria-busy', String(sending));
newsletterSpinner.hidden = !sending;
newsletterSubmitLabel.textContent = sending ? 'Subscribing…' : 'Subscribe';
newsletterForm.classList.toggle('is-sending', sending);
newsletterEmail.readOnly = sending;
newsletterConsent.disabled = sending;
document.getElementById('newsletter-back').disabled = sending;
}
const newsletterStatus = document.getElementById('newsletter-status');
const honeypot = document.getElementById('newsletter-website');
const newsletterSuccess = document.getElementById('newsletter-success');
const successTitle = document.getElementById('success-title');
const successText = document.getElementById('success-text');
// Replace the form with a clear confirmation, mentioning the email that was saved
function showSignupSuccess(email, alreadySubscribed) {
newsletterForm.reset();
newsletterForm.hidden = true;
if (alreadySubscribed) {
successTitle.textContent = "You're already subscribed";
successText.textContent = `${email} is already on our list. We'll let you know when Else launches.`;
} else {
successTitle.textContent = "You're subscribed!";
successText.textContent = `Thanks for joining. We'll email you at ${email} when Else launches.`;
}
newsletterSuccess.hidden = false;
document.getElementById('success-back').focus();
}
// Newsletter config.
// NEWSLETTER_ENDPOINT: Web app URL of the Google Apps Script that saves signups and visits
// to Google Sheets (see newsletter-apps-script.gs). Leave empty to disable both.
const NEWSLETTER_ENDPOINT = 'https://script.google.com/macros/s/AKfycbzZ4DXSE1MHEBm0YE2ja69Lr7MSzBOmjRJs6lUyJSHpe6pwrZc6FRbVRC4Y0FPeDphN/exec';
// Count this visit anonymously: page, referring site, device type and time zone only.
// No cookies or browser storage are used, so no cookie banner is needed.
function recordVisit() {
if (!NEWSLETTER_ENDPOINT || !navigator.sendBeacon) return;
let referrer = '';
try {
const host = document.referrer ? new URL(document.referrer).hostname : '';
if (host && host !== location.hostname) referrer = host;
} catch {
// Ignore malformed referrers
}
const visit = new URLSearchParams({
type: 'visit',
page: location.pathname,
referrer: referrer || 'direct',
device: window.matchMedia('(max-width: 600px)').matches ? 'mobile' : 'desktop',
// Approximate region from the device's time zone (e.g. "Europe/Madrid"); no IP lookup
timezone: Intl.DateTimeFormat().resolvedOptions().timeZone || 'unknown',
});
navigator.sendBeacon(NEWSLETTER_ENDPOINT, visit);
}
recordVisit();
// Bump this date whenever the consent text changes, so each signup records which version was accepted
const CONSENT_VERSION = '2026-10-21';
// Collapse the signup form back to the single button
function resetNewsletter() {
footerText.hidden = false;
aboutButton.hidden = false;
newsletterForm.hidden = true;
newsletterSuccess.hidden = true;
newsletterButton.hidden = false;
newsletterButton.setAttribute('aria-expanded', 'false');
newsletterStatus.textContent = '';
}
// Restart the cascade animation on the given elements
function replayCascade(elements) {
document.body.classList.add('is-replay');
elements.forEach(el => {
el.classList.remove('reveal');
void el.offsetWidth; // force reflow so the animation restarts
el.classList.add('reveal');
});
}
// Crossfade to the "ee" logo, show contact details and replay the cascade
aboutButton.addEventListener('click', () => {
aboutButton.hidden = true;
newsletter.hidden = true;
homeView.classList.add('is-hidden');
aboutView.classList.remove('is-hidden');
contactInfo.hidden = false;
replayCascade([...aboutView.querySelectorAll('.reveal'), contactInfo, footerText]);
});
// Crossfade back to the "ELSE" logo and replay the cascade
backButton.addEventListener('click', () => {
contactInfo.hidden = true;
wipNote.textContent = '';
aboutView.classList.add('is-hidden');
homeView.classList.remove('is-hidden');
aboutButton.hidden = false;
resetNewsletter();
newsletter.hidden = false;
replayCascade([...homeView.querySelectorAll('.reveal'), footerText, aboutButton, newsletter]);
});
// Open the signup form. It takes the place of the tagline and About button
// so the footer doesn't grow into the logo on short screens.
newsletterButton.addEventListener('click', () => {
footerText.hidden = true;
aboutButton.hidden = true;
newsletterButton.hidden = true;
newsletterButton.setAttribute('aria-expanded', 'true');
newsletterForm.hidden = false;
replayCascade([newsletter]);
newsletterEmail.focus();
});
// Close the signup form or confirmation and bring back the home footer
['newsletter-back', 'success-back'].forEach(id => {
document.getElementById(id).addEventListener('click', () => {
resetNewsletter();
replayCascade([footerText, aboutButton, newsletter]);
newsletterButton.focus();
});
});
// Validate and send the signup
newsletterForm.addEventListener('submit', async event => {
event.preventDefault();
if (isSending) return; // ignore extra clicks or Enter presses while sending
newsletterStatus.textContent = '';
// Bots fill the hidden field: pretend it worked and send nothing
if (honeypot.value) {
showSignupSuccess(newsletterEmail.value.trim(), false);
return;
}
if (!newsletterEmail.checkValidity()) {
newsletterStatus.textContent = 'Please enter a valid email address.';
newsletterEmail.focus();
return;
}
// Consent must be an explicit, active choice
if (!newsletterConsent.checked) {
newsletterStatus.textContent = 'Please tick the box to agree to receive the newsletter.';
newsletterConsent.focus();
return;
}
if (!NEWSLETTER_ENDPOINT) {
newsletterStatus.textContent = 'Signups open very soon. Thanks for your interest!';
return;
}
// Send the email plus a record of what was consented to and when
// Sent as a plain form so the browser doesn't need a CORS preflight
const data = new URLSearchParams();
data.append('email', newsletterEmail.value.trim());
data.append('consent', 'yes');
data.append('consent_version', CONSENT_VERSION);
data.append('consent_text', document.getElementById('consent-text').textContent.replace(/\s+/g, ' ').trim());
data.append('consent_timestamp', new Date().toISOString());
setSending(true);
try {
const response = await fetch(NEWSLETTER_ENDPOINT, { method: 'POST', body: data });
if (!response.ok) throw new Error(`HTTP ${response.status}`);
// Read as text first so a non-JSON reply (e.g. a Google sign-in page) shows up in the console
const raw = await response.text();
let result;
try {
result = JSON.parse(raw);
} catch {
throw new Error(`Endpoint did not return JSON. Check the Web app access settings. Response starts with: ${raw.slice(0, 200)}`);
}
if (!result.ok) throw new Error(result.error || 'Signup rejected');
showSignupSuccess(data.get('email'), result.duplicate === true);
} catch (error) {
console.error('Newsletter signup failed:', error);
newsletterStatus.textContent = 'Something went wrong. Please try again later.';
} finally {
setSending(false);
}
});
// Links not ready yet: show an inline work-in-progress notice
document.querySelectorAll('.wip').forEach(link => {
link.addEventListener('click', event => {
event.preventDefault();
wipNote.textContent = 'Work in progress';
});
});
</script>
</body>
